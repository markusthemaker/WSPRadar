SELECT id, time, band, rx_sign, rx_loc, tx_sign, tx_loc, frequency, power, snr, distance FROM wspr.rx WHERE band=7 AND time >= toDateTime('2010-12-19 12:00:00') AND time <= toDateTime('2010-12-20 20:00:00') AND ((tx_sign='VE6PDQ' AND rx_sign IN ('KP4MD','WB6RQN')) OR (rx_sign='VE6PDQ' AND tx_sign IN ('KP4MD','WB6RQN'))) ORDER BY time, id FORMAT JSON
